import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid>
  <Card title="An agent writes it" icon="pencil">
    A deck is typed JSON with stable ids. The MCP server gives an agent seven tools to read, edit, comment on, render,
    present, export and add images to it. An edit is one transaction against a revision, and a stale revision is
    rejected, never merged.
  </Card>
  <Card title="A person repairs it" icon="setting">
    The browser editor sends its changes through the same operations an agent uses. It saves by itself, and a draft
    survives a closed tab. You can pin any block anywhere on the slide, comment on it, and accept or revert what an
    agent changed.
  </Card>
  <Card title="Both present it" icon="laptop">
    A presenter window with notes, the next click and two clocks, a reading view for the audience, and a control lease: hand the deck
    to an agent, and take it back at once.
  </Card>
  <Card title="It works offline" icon="download">
    `depthcue export` writes a folder that opens by double-click, with no server and no network.
  </Card>
</CardGrid>
